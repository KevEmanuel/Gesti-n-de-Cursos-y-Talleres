Nota: la configuración activa está en app/config/database.php. Los valores por defecto deben corresponder al servidor y la base de datos local; aquí se muestra la configuración utilizada pero debera usar segun las definidas en su maquina local
tanto para usuario como la contraseña que tenga utiliza en Xammp

<?php

declare(strict_types=1);

// Crea una conexion PDO segura usando la configuracion local
function getDatabaseConnection(): PDO
{
    $host = getenv('DB_HOST') ?: '127.0.0.1';
    $database = getenv('DB_NAME') ?: 'gestiondedatos';
    $username = getenv('DB_USER') ?: 'root';
    $password = getenv('DB_PASSWORD') ?: '';

    $dsn = "mysql:host={$host};dbname={$database};charset=utf8mb4";

    return new PDO($dsn, $username, $password, [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES => false,
    ]);
}
